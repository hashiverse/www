1//! # Inbound RPC dispatch loop
2//!
3//! The hot path of the server: a single async loop that drains `IncomingRequest`s
4//! from the transport's `mpsc::Receiver`, decodes each packet via
5//! [`hashiverse_lib::protocol::rpc::rpc_request::RpcRequestPacketRx`], and routes
6//! by [`hashiverse_lib::protocol::payload::payload::PayloadRequestKind`] to the
7//! correct per-op handler (bootstrap, announce, get/submit/heal/cache post
8//! bundles, get/submit/heal/cache feedback, fetch URL preview, trending
9//! hashtags, ping).
10//!
11//! Per-request safety checks happen before any real work:
12//!
13//! - **PoW verification** — the packet's PoW must be sufficient for *this* server's
14//!   identity. Anything under-powered or stale is dropped immediately.
15//! - **Replay protection** — a short-lived salt cache rejects salts we've already
16//!   seen, so a valid signed request can't be replayed from another network vantage.
17//! - **Peer upgrade** — if the caller's embedded [`hashiverse_lib::protocol::peer::Peer`] carries a stronger PoW
18//!   than what we have in the tracker, the tracker is upgraded in place.
19//!
20//! The loop respects a `CancellationToken` so graceful shutdown drains in-flight
21//! work and stops accepting new requests cleanly.

59/// Fallback hashtags used to top up a trending-hashtags response when the server
60/// does not yet know enough real trending hashtags to satisfy the requested limit.
61/// Applied in order, skipping any entry whose normalised form is already present
62/// in the real trending list. Filler entries are returned with `count = 0` so
63/// clients can distinguish seeded fillers from genuine trending data.

66/// Normalise a hashtag for equality comparison: lowercase, with any leading `#`
67/// stripped. Mirrors the canonicalisation performed by `Id::from_hashtag_str`.
68fn normalise_hashtag(hashtag: &str) -> String {
69    let lowercased = hashtag.to_lowercase();
70    match lowercased.strip_prefix('#') {
71        Some(stripped) => stripped.to_string(),
72        None => lowercased,
73    }
74}

76/// Top up `trending_hashtags` from `fallback_hashtags` (in order) until it reaches
77/// `limit`, skipping any fallback whose normalised form already appears in the list.
78/// Filler entries are inserted with `count = 0`. No-op if the list is already at
79/// or above the limit.
80/// Is `donor_id` close enough to `location_id` that it plausibly is (or was) one of the servers responsible for that location?
81///
82/// `nearest_peer_ids` is our own view of the nearest servers to the location (as returned by `get_peers_for_key(location_id, required_count)`,
83/// which may include ties beyond `required_count`).  The donor must be no more than `HEAL_DONOR_LEADING_AGREEMENT_BITS_SLACK` bits further out
84/// than the furthest of those.  Deliberately a distance test rather than "is the donor in our view": the donor may have died long ago.
85/// If we do not yet know `required_count` servers we cannot rule anyone out.
86fn is_plausible_heal_donor(donor_id: &Id, location_id: &Id, nearest_peer_ids: &[Id], required_count: usize) -> bool {
87    if nearest_peer_ids.len() < required_count {
88        return true;
89    }
90
91    let furthest_nearest_leading_agreement_bits = nearest_peer_ids
92        .iter()
93        .map(|nearest_peer_id| tools::leading_agreement_bits_xor(location_id.as_ref(), nearest_peer_id.as_ref()))
94        .min()
95        .unwrap_or(LeadingAgreementBits::MIN);
96    let donor_leading_agreement_bits = tools::leading_agreement_bits_xor(location_id.as_ref(), donor_id.as_ref());
97
98    donor_leading_agreement_bits >= furthest_nearest_leading_agreement_bits - config::HEAL_DONOR_LEADING_AGREEMENT_BITS_SLACK
99}

101fn top_up_trending_hashtags_with_fallback(trending_hashtags: &mut Vec<TrendingHashtagV1>, limit: u16, fallback_hashtags: &[&str]) {
102    let target_length = limit as usize;
103    if trending_hashtags.len() >= target_length {
104        return;
105    }
106
107    let mut existing_normalised_hashtags: HashSet<String> = trending_hashtags.iter()
108        .map(|entry| normalise_hashtag(&entry.hashtag))
109        .collect();
110
111    for fallback_hashtag in fallback_hashtags {
112        if trending_hashtags.len() >= target_length {
113            break;
114        }
115        let normalised_fallback_hashtag = normalise_hashtag(fallback_hashtag);
116        if existing_normalised_hashtags.contains(&normalised_fallback_hashtag) {
117            continue;
118        }
119        trending_hashtags.push(TrendingHashtagV1 {
120            hashtag: (*fallback_hashtag).to_string(),
121            count: 0,
122        });
123        existing_normalised_hashtags.insert(normalised_fallback_hashtag);
124    }
125}

127/// Awaits a DNS lookup for a URL preview host, giving up after `dns_timeout`.  The dispatch loop is serial, so an
128/// unbounded lookup here would stall every other request.  `lookup` is injected so the timeout can be tested without a network.
129async fn resolve_url_preview_host<I>(host: &str, dns_timeout: std::time::Duration, lookup: impl Future<Output = std::io::Result<I>>) -> anyhow::Result<Vec<std::net::SocketAddr>>
130where
131    I: Iterator<Item = std::net::SocketAddr>,
132{
133    match tokio::time::timeout(dns_timeout, lookup).await {
134        Err(_) => anyhow::bail!("FetchUrlPreviewV1: DNS resolution timed out for {}", host),
135        Ok(Err(e)) => anyhow::bail!("FetchUrlPreviewV1 SSRF: DNS resolution failed for {}: {}", host, e),
136        Ok(Ok(socket_addrs)) => Ok(socket_addrs.collect()),
137    }
138}

140/// Awaits `future`, converting a panic inside it into an `Err` so that a single bad
141/// request cannot unwind through the dispatch loop and stop the server.
142async fn catch_panic_as_error<T>(future: impl Future<Output = T>) -> anyhow::Result<T> {
143    match AssertUnwindSafe(future).catch_unwind().await {
144        Ok(value) => Ok(value),
145        Err(panic_payload) => {
146            let panic_message = if let Some(panic_message) = panic_payload.downcast_ref::<&str>() {
147                panic_message.to_string()
148            } else if let Some(panic_message) = panic_payload.downcast_ref::<String>() {
149                panic_message.clone()
150            } else {
151                "<non-string panic payload>".to_string()
152            };
153            Err(anyhow::anyhow!("panic: {}", panic_message))
154        }
155    }
156}

158/// Replies to (or abandons) `incoming` according to how dispatching it went.
159/// The outer `Err` is a handler panic (see [`catch_panic_as_error`]); it is scored as a bad request just like
160/// an ordinary `Err`, so a caller who has found a panic can't keep triggering it without being DDoS-scored.
161fn complete_incoming_request(incoming: IncomingRequest, dispatch_outcome: anyhow::Result<anyhow::Result<BytesGatherer>>) {
162    match dispatch_outcome {
163        Ok(Ok(bytes)) => {
164            let result = incoming.reply.send(bytes);
165            if result.is_err() { warn!("failed to send reply"); }
166        },
167        Err(handler_panic) => {
168            error!("handler panicked processing packet from {}: {}", incoming.caller_address, handler_panic);
169            incoming.report_bad_request();
170            drop(incoming.reply);
171        },
172        Ok(Err(e)) => {
173            warn!("failed to process packet from {}: {}", incoming.caller_address, e);
174            incoming.report_bad_request();
175            drop(incoming.reply);
176        },
177    }
178}

181    pub async fn wrap_and_dispatch_network_envelopes(&self, cancellation_token: CancellationToken, mut rx: mpsc::Receiver<IncomingRequest>) -> Result<(), anyhow::Error> {
182        loop {
183            tokio::select! {
184                _ = cancellation_token.cancelled() => { break },
185
186                receipt = rx.recv() => {
187                    match receipt {
188                        Some(incoming) => {
189                            // trace!("dispatch_network_envelopes received bytes={:?}", incoming.bytes);
190                            // A panic in any handler must not unwind out of this loop: it runs inside the server's
191                            // try_join! and would take the whole server down with it.
192                            let dispatch_outcome = catch_panic_as_error(self.wrap_and_dispatch_network_envelope(cancellation_token.clone(), &incoming)).await;
193                            complete_incoming_request(incoming, dispatch_outcome);
194                        },
195                        None => {
196                            warn!("channel closed");
197                            break;
198                        }
199                    }
200                }
201            }
202        }
203
204        Ok(())
205    }

207    async fn wrap_and_dispatch_network_envelope(&self, cancellation_token: CancellationToken, incoming: &IncomingRequest) -> anyhow::Result<BytesGatherer> {
208        let caller_address = incoming.caller_address.as_str();
209        let current_time_millis = self.runtime_services.time_provider.current_time_millis();
210
211        // Decode the envelope
212        let rpc_request_packet_rx = RpcRequestPacketRx::decode(&current_time_millis, &self.server_id.keys.verification_key_bytes, &self.server_id.keys.pq_commitment_bytes, incoming.bytes.clone())?;
213        // trace!("payload_request_kind={}", rpc_request_packet_rx.payload_request_kind);
214
215        // Count this inbound request. After decoding (so malformed traffic doesn't
216        // pollute the totals) but before the per-handler PoW check (so failed-PoW
217        // attempts still show up — adversarial load is the load we'd want to see).
218        let payload_request_kind_index = rpc_request_packet_rx.payload_request_kind.clone() as usize;
219        self.request_counters[payload_request_kind_index].fetch_add(1, Ordering::Relaxed);
220        // Also feed the decaying per-window rate estimates so the stats reflect
221        // recent load, not just the monotonic all-time total.
222        self.request_rate_windows[payload_request_kind_index].lock().record(current_time_millis);
223
224        // Check that we have not seen this salt recently (stops replay attacks)
225        {
226            if self.seen_salts.contains_key(&rpc_request_packet_rx.pow_salt) {
227                anyhow::bail!("replay detected: salt already seen");
228            }
229            self.seen_salts.insert(rpc_request_packet_rx.pow_salt, ());
230        }
231
232        // Keep this for our response
233        let pow_content_hash = rpc_request_packet_rx.pow_content_hash;
234
235        let dispatch_result: anyhow::Result<BytesGatherer> = try {
236            // Check that the pow is meaningful
237            let pow = match rpc_request_packet_rx.pow_server_known {
238                true => {
239                    let (pow, improved_pow_current_day, improved_pow_current_month) = {
240                        let peer_self = self.peer_self.read(); // Remember alphabetical locking order!
241                        let pow = PeerPow::new(
242                            rpc_request_packet_rx.pow_sponsor_id,
243                            &peer_self.verification_key_bytes,
244                            &peer_self.pq_commitment_bytes,
245                            rpc_request_packet_rx.pow_timestamp,
246                            rpc_request_packet_rx.pow_content_hash,
247                            rpc_request_packet_rx.pow_salt,
248                        )?;
249
250                        let improved_pow_current_day = pow.pow_decayed_day(current_time_millis) > peer_self.pow_current_day.pow_decayed_day(current_time_millis);
251                        let improved_pow_current_month = pow.pow_decayed_month(current_time_millis) > peer_self.pow_current_month.pow_decayed_month(current_time_millis);
252
253                        (pow, improved_pow_current_day, improved_pow_current_month)
254                    };
255
256                    // Check if we need to modify peer_self
257                    if improved_pow_current_day || improved_pow_current_month {
258                        let mut peer_self = self.peer_self.write(); // Remember alphabetical locking order!
259                        if improved_pow_current_day {
260                            trace!("pow_current_day upgraded {} -> {}", peer_self.pow_current_day, pow);
261                            peer_self.pow_current_day = pow.clone();
262                        }
263                        if improved_pow_current_month {
264                            trace!("pow_current_month upgraded {} -> {}", peer_self.pow_current_month, pow);
265                            peer_self.pow_current_month = pow.clone();
266                        }
267
268                        peer_self.sign(self.runtime_services.time_provider.as_ref(), &self.server_id.keys.signature_key)?;
269                    }
270
271                    Some(pow)
272                }
273
274                false => {
275                    // Only some request types are allowed anonymous pow
276                    match rpc_request_packet_rx.payload_request_kind {
277                        PayloadRequestKind::BootstrapV1 => {}
278                        _ => anyhow::bail!("Anonymous pow not allowed for {}", rpc_request_packet_rx.payload_request_kind),
279                    }
280
281                    None
282                }
283            };
284
285            // Dispatch appropriately
286            let (compress_response, payload_response_kind, payload) = self.dispatch_network_envelope(cancellation_token, pow, rpc_request_packet_rx).await?;
287            let response_flags = match compress_response {
288                true => RpcResponsePacketTxFlags::COMPRESSED,
289                false => RpcResponsePacketTxFlags::empty(),
290            };
291
292            // Encode response
293            RpcResponsePacketTx::encode(
294                &self.server_id.keys.signature_key,
295                &self.server_id.keys.verification_key_bytes,
296                &self.server_id.keys.pq_commitment_bytes,
297                &self.server_id.sponsor_id,
298                &self.server_id.timestamp,
299                &self.server_id.hash,
300                &self.server_id.salt,
301                &pow_content_hash,
302                response_flags,
303                payload_response_kind,
304                payload,
305            )?
306        };
307
308        match dispatch_result {
309            Ok(results) => Ok(results),
310            Err(e) => {
311                warn!("failed to dispatch packet from {}: {}", caller_address, e);
312                incoming.report_bad_request();
313
314                let payload_response_kind = PayloadResponseKind::ErrorResponseV1;
315                let response = ErrorResponseV1 { code: 0, message: e.to_string() };
316                let payload = BytesGatherer::from_bytes(json::struct_to_bytes(&response)?);
317
318                // Encode response
319                RpcResponsePacketTx::encode(
320                    &self.server_id.keys.signature_key,
321                    &self.server_id.keys.verification_key_bytes,
322                    &self.server_id.keys.pq_commitment_bytes,
323                    &self.server_id.sponsor_id,
324                    &self.server_id.timestamp,
325                    &self.server_id.hash,
326                    &self.server_id.salt,
327                    &pow_content_hash,
328                    RpcResponsePacketTxFlags::COMPRESSED,
329                    payload_response_kind,
330                    payload,
331                )
332            }
333        }
334    }

336    async fn dispatch_network_envelope(&self, cancellation_token: CancellationToken, pow: Option<PeerPow>, rpc_request_packet_rx: RpcRequestPacketRx) -> anyhow::Result<(bool, PayloadResponseKind, BytesGatherer)> {
337        // Where do we want to decide if we should compress?  Here in one block, or at the end of each individual dispatch_xxx?
338        let compress_response = match rpc_request_packet_rx.payload_request_kind {
339            PayloadRequestKind::GetPostBundleV1 => false,   // We don't compress these again as they are already predominantly compressed
340            PayloadRequestKind::CachePostBundleV1 => false, // We don't compress these again as they are already predominantly compressed
341            _ => true,
342        };
343
344        let (payload_response_kind, payload) = match rpc_request_packet_rx.payload_request_kind {
345            PayloadRequestKind::ErrorV1 => {
346                anyhow::bail!("Received ErrorV1");
347            }
348            PayloadRequestKind::PingV1 => self.dispatch_network_payload_x_PingV1(cancellation_token, rpc_request_packet_rx.payload_request_kind, rpc_request_packet_rx.bytes).await?,
349            PayloadRequestKind::BootstrapV1 => self.dispatch_network_payload_x_BootstrapV1(cancellation_token, rpc_request_packet_rx.payload_request_kind, rpc_request_packet_rx.bytes).await?,

352            }
353            PayloadRequestKind::GetPostBundleV1 => self.dispatch_network_payload_x_GetPostBundleV1(cancellation_token, rpc_request_packet_rx.payload_request_kind, rpc_request_packet_rx.bytes).await?,
354            PayloadRequestKind::GetPostBundleFeedbackV1 => { self.dispatch_network_payload_x_GetPostBundleFeedbackV1(cancellation_token, rpc_request_packet_rx.payload_request_kind, rpc_request_packet_rx.bytes).await? }
355            PayloadRequestKind::SubmitPostClaimV1 => { self.dispatch_network_payload_x_SubmitPostClaimV1(cancellation_token, pow, rpc_request_packet_rx.payload_request_kind, rpc_request_packet_rx.bytes).await? }

359            PayloadRequestKind::HealPostBundleCommitV1 => { self.dispatch_network_payload_x_HealPostBundleCommitV1(cancellation_token, rpc_request_packet_rx.payload_request_kind, rpc_request_packet_rx.bytes).await? }

361            PayloadRequestKind::CachePostBundleV1 => self.dispatch_network_payload_x_CachePostBundleV1(cancellation_token, rpc_request_packet_rx.payload_request_kind, rpc_request_packet_rx.bytes).await?,
362            PayloadRequestKind::CachePostBundleFeedbackV1 => { self.dispatch_network_payload_x_CachePostBundleFeedbackV1(cancellation_token, rpc_request_packet_rx.payload_request_kind, rpc_request_packet_rx.bytes).await? }
363            PayloadRequestKind::FetchUrlPreviewV1 => self.dispatch_network_payload_x_FetchUrlPreviewV1(cancellation_token, pow, rpc_request_packet_rx.payload_request_kind, rpc_request_packet_rx.bytes).await?,
364            PayloadRequestKind::TrendingHashtagsFetchV1 => self.dispatch_network_payload_x_TrendingHashtagsFetchV1(cancellation_token, rpc_request_packet_rx.payload_request_kind, rpc_request_packet_rx.bytes).await?,

367        };

370    }

389    async fn dispatch_network_payload_x_AnnounceV2(&self, _cancellation_token: CancellationToken, payload_request_kind: PayloadRequestKind, bytes: Bytes) -> anyhow::Result<(PayloadResponseKind, BytesGatherer)> {
390        anyhow_assert_eq!(&PayloadRequestKind::AnnounceV2, &payload_request_kind);
391
392        let request = json::bytes_to_struct::<AnnounceV2>(&bytes)?;
393
394        let peer = request.peer_self;
395        let peer_id = peer.id;
396        let now = self.runtime_services.time_provider.as_ref().current_time_millis();
397
398        // Gate Kademlia admission on the transport-specific ownership proof. For HTTPS that's
399        // an offline X.509 chain validation against bundled webpki-roots with an IP-SAN match;
400        // for mem / TCP it's the empty-marker check. A failed proof is a hard error: no
401        // Kademlia add, no nearest-peers list returned. The dispatcher's Err -> ErrorResponseV1
402        // wrapper also reports the request as bad to the DDoS layer, so repeat offenders get
403        // throttled by per-IP scoring.
404        let ownership_proof = self.transport_server.get_transport_ownership_proof();
405        if !ownership_proof.prove(&peer, &request.proof_payload, now) {
406            anyhow::bail!("AnnounceV2 from peer {} at {} failed ownership proof", peer_id, peer.address);
407        }
408        self.add_potential_peer_to_kademlia(peer, now).await;
409
410        let (peers_nearest, _) = self.kademlia.read().get_peers_for_key(&peer_id, config::ANNOUNCE_NUM_PEERS);
411
412        let json = json::struct_to_bytes(&AnnounceResponseV2 {
413            peer_self: self.peer_self.read().clone(),
414            peers_nearest,
415        })?;
416        Ok((PayloadResponseKind::AnnounceResponseV2, BytesGatherer::from_bytes(json)))
417    }

1015    fn commit_heal_post_bundle(&self, request: &HealPostBundleCommitV1, peer_self: &Peer, time_millis: TimeMillis) -> anyhow::Result<bool> {

1017        if request.donor_header.signature != request.token.donor_header_signature {

1021

1023            anyhow::bail!("HealPostBundleCommitV1: token location_id does not match donor_header");

1027
1028        // Parse the post bytes supplied by the client (one entry per token.needed_post_id, in order)
1029        let mut remaining_bytes = request.encoded_posts_bytes.clone();
1030        let mut posts_to_add: Vec<(Id, Bytes)> = Vec::new();

1034                .encoded_post_ids
1035                .iter()
1036                .zip(request.donor_header.encoded_post_lengths.iter())
1037                .find(|(id, _)| *id == post_id)
1038                .map(|(_, len)| *len)
1039                .ok_or_else(|| anyhow::anyhow!("needed_post_id {} not found in donor_header", post_id))?;

1041                anyhow::bail!("HealPostBundleCommitV1: not enough bytes for post {}", post_id);
1042            }

1046        if !remaining_bytes.is_empty() {
1047            anyhow::bail!("HealPostBundleCommitV1: {} excess bytes", remaining_bytes.len());
1048        }

1051        // and pass the same admission rules we would have applied had it been submitted to us directly.
1052        // The reference time is the earlier of the donor's time and ours: the donor header may itself be up to the threshold ahead of
1053        // our clock, and using it alone would let healed posts sit twice the threshold in the future.
1054        let admission_reference_time_millis = std::cmp::min(request.donor_header.time_millis, time_millis);
1055        for (post_id, post_bytes) in &posts_to_add {
1056            let decoded_post = EncodedPostV1::decode_from_bytes(post_bytes.clone(), &request.token.bucket_location.base_id, true, true)

1058            if decoded_post.post_id != *post_id {
1059                anyhow::bail!("HealPostBundleCommitV1: post bytes are for {} not {}", decoded_post.post_id, post_id);
1060            }
1061            decoded_post
1062                .check_admissible_to_bucket(&request.token.bucket_location, admission_reference_time_millis)
1063                .map_err(|e| anyhow::anyhow!("HealPostBundleCommitV1: {}", e))?;
1064        }
1065
1066        // Load our current bundle (or start empty)
1067        let _lock = self.environment.get_write_lock_for_location_id(&location_id);
1068        let post_bundle_bytes = self.environment.get_post_bundle_bytes(time_millis, &location_id)?;
1069        let post_bundle_metadata = self.environment.get_post_bundle_metadata(time_millis, &location_id)?;
1070        let mut post_bundle_metadata = post_bundle_metadata.unwrap_or_else(PostBundleMetadata::zero);
1071
1072        let mut post_bundle = match post_bundle_bytes {
1073            Some(b) => EncodedPostBundleV1::from_bytes(Bytes::from_owner(b), true)?,
1074            None => EncodedPostBundleV1 {
1075                header: EncodedPostBundleHeaderV1 {
1076                    time_millis: TimeMillis::zero(),
1077                    location_id,
1078                    overflowed: false, // Never adopt the donor's flags - they are recomputed from our own state below
1079                    sealed: false,
1080                    num_posts: 0,
1081                    encoded_post_ids: vec![],
1082                    encoded_post_lengths: vec![],
1083                    encoded_post_healed: HashSet::new(),
1084                    peer: peer_self.clone(),
1085                    signature: Signature::zero(),
1086                },
1087                encoded_posts_bytes: Bytes::new(),
1088            },
1089        };
1090
1091        let our_post_ids: HashSet<Id> = post_bundle.header.encoded_post_ids.iter().copied().collect();
1092
1093        let mut posts_mut = BytesMut::from(post_bundle.encoded_posts_bytes.as_ref());
1094        let mut added_any = false;
1095        for (post_id, post_bytes) in posts_to_add {
1096            if post_bundle.header.encoded_post_ids.len() >= u8::MAX as usize {
1097                warn!("HealPostBundleCommitV1: bundle for location_id={} is full so not healing any more posts into it", location_id);
1098                break;
1099            }
1100            if !our_post_ids.contains(&post_id) {
1101                let len = post_bytes.len();
1102                posts_mut.extend_from_slice(&post_bytes);
1103                post_bundle.header.encoded_post_ids.push(post_id);
1104                post_bundle.header.encoded_post_lengths.push(len);
1105                post_bundle.header.encoded_post_healed.insert(post_id);
1106                added_any = true;
1107            }
1108        }
1109        post_bundle.encoded_posts_bytes = posts_mut.freeze();
1110
1111        if added_any {
1112            post_bundle.header.time_millis = time_millis;
1113            post_bundle.header.num_posts = post_bundle.header.encoded_post_ids.len() as u8;
1114            post_bundle.header.overflowed = post_bundle.header.num_posts > config::ENCODED_POST_BUNDLE_V1_OVERFLOWED_NUM_POSTS || post_bundle_metadata.num_posts_granted > config::ENCODED_POST_BUNDLE_V1_OVERFLOWED_NUM_POSTS_GRANTED;
1115            post_bundle.header.sealed = request.token.bucket_location.is_sealed_at(time_millis, post_bundle.header.overflowed);
1116            post_bundle.header.signature_generate(&self.server_id.keys.signature_key)?;
1117
1118            let new_bytes = post_bundle.to_bytes()?;
1119            post_bundle_metadata.num_posts = post_bundle.header.num_posts;
1120            post_bundle_metadata.size = post_bundle.header.encoded_post_lengths.iter().sum();
1121            post_bundle_metadata.overflowed = post_bundle.header.overflowed;
1122            post_bundle_metadata.sealed = post_bundle.header.sealed;
1123
1124            self.environment.put_post_bundle_bytes(time_millis, &location_id, &new_bytes)?;
1125            self.environment.put_post_bundle_metadata(time_millis, &location_id, &post_bundle_metadata, 0)?;
1126
1127            info!("Healed {} post(s) for location_id={}", post_bundle.header.encoded_post_healed.len(), location_id);
1128        }
1129
1130        Ok(added_any)
1131    }

1288        let host = if host_and_port.starts_with('[') {
1289            // IPv6 literal [addr] or [addr]:port
1290            host_and_port.trim_start_matches('[').split(']').next().unwrap_or("")
1291        } else {
1292            // hostname or IPv4 — strip optional port
1293            host_and_port.split(':').next().unwrap_or(host_and_port)
1294        };
1295        if host.is_empty() {
1296            anyhow::bail!("FetchUrlPreviewV1 SSRF: could not extract host from URL");
1297        }
1298        if host.parse::<std::net::IpAddr>().is_ok() {
1299            anyhow::bail!("FetchUrlPreviewV1 SSRF: bare IP addresses are not allowed");
1300        }
1301
1302        // Resolve once and validate every returned address.  Collecting into a Vec lets us
1303        // re-use the same addresses to pin the reqwest client, closing the DNS-rebinding window
1304        // (nip.io, metadata.google.internal, TTL=0 rebind, etc.).
1305        let resolved_socket_addrs: Vec<std::net::SocketAddr> = resolve_url_preview_host(host, config::URL_FETCH_DNS_TIMEOUT_MILLIS.into(), tokio::net::lookup_host((host, 443u16))).await?;
1306        if resolved_socket_addrs.is_empty() {
1307            anyhow::bail!("FetchUrlPreviewV1 SSRF: DNS returned no addresses for {}", host);
1308        }
1309        for socket_addr in &resolved_socket_addrs {
1310            let ip = socket_addr.ip();
1311            if is_ssrf_protected_ip(ip) {
1312                anyhow::bail!("FetchUrlPreviewV1 SSRF: {} resolved to protected address {}", host, ip);
1313            }
1314        }
1315
1316        // Build a client that:
1317        //   - resolve_to_addrs: skips re-resolution entirely — the validated IPs are used directly
1318        //   - redirect::none: prevents a server-side redirect to an unvalidated internal URL
1319        //   - no_proxy: ignores HTTP_PROXY / NO_PROXY env vars that could route to internal hosts
1320        //   - times out quickly in unreasonable scenarios
1321        //   - limits download size
1322        let http_client = reqwest::Client::builder()
1323            .connect_timeout(config::URL_FETCH_CONNECT_TIMEOUT_MILLIS.into())
1324            .timeout(config::URL_FETCH_TOTAL_TIMEOUT_MILLIS.into())
1325            .user_agent("hashiverse-preview/1.0")
1326            .resolve_to_addrs(host, &resolved_socket_addrs)
1327            .redirect(reqwest::redirect::Policy::none())
1328            .no_proxy()
1329            .build()?;
1330
1331        const URL_FETCH_MAX_BODY_BYTES: usize = 2 * 1024 * 1024;
1332        let mut http_response = http_client.get(&request.url).send().await?;
1333
1334        // Reject early if Content-Length already exceeds the limit — before reading any body bytes.
1335        if let Some(content_length) = http_response.content_length() {
1336            if content_length > URL_FETCH_MAX_BODY_BYTES as u64 {
1337                anyhow::bail!("FetchUrlPreviewV1: Content-Length {} exceeds {} byte limit", content_length, URL_FETCH_MAX_BODY_BYTES);
1338            }
1339        }
1340        let mut body_bytes = BytesMut::new();
1341        while let Some(chunk) = http_response.chunk().await? {
1342            // body_bytes.len() is already within the limit; only the new chunk can overflow.
1343            // We copy only as many bytes as fit rather than bailing, so a single large chunk
1344            // (which reqwest has already buffered) doesn't cause an error — we just truncate.
1345            // This is sufficient for HTML preview extraction which only needs the <head>.
1346            let remaining = URL_FETCH_MAX_BODY_BYTES - body_bytes.len();
1347            body_bytes.extend_from_slice(&chunk[..chunk.len().min(remaining)]);
1348            if body_bytes.len() >= URL_FETCH_MAX_BODY_BYTES {
1349                break;
1350            }
1351        }
1352        let html = String::from_utf8_lossy(&body_bytes).into_owned();
1353
1354        let preview_data = url_preview::extract_url_preview(&html);
1355
1356        let response = FetchUrlPreviewResponseV1 {
1357            url: if preview_data.canonical_url.is_empty() { request.url } else { preview_data.canonical_url },
1358            title: preview_data.title,
1359            description: preview_data.description,
1360            image_url: preview_data.image_url,
1361        };
1362
1363        Ok((PayloadResponseKind::FetchUrlPreviewResponseV1, BytesGatherer::from_bytes(response.to_bytes()?)))
1364    }
