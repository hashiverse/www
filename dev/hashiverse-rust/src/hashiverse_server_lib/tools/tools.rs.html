1//! # Server-only helpers
2//!
3//! A small grab bag of utilities the server needs that don't belong in
4//! `hashiverse-lib`:
5//!
6//! - [`get_public_ipv4`] — asks ipify.org for the node's externally-visible IPv4 so
7//!   the node's advertised peer address matches reality (behind NAT / firewalls).
8//!   Falls back to loopback when explicitly running in a local-only test network.
9//! - [`is_ssrf_protected_ip`] — the deny-list for every server-initiated outbound
10//!   fetch (URL previews, Let's Encrypt challenges, ipify). Rejects loopback,
11//!   RFC-1918 private ranges, link-local, CGNAT, cloud-metadata endpoints
12//!   (169.254.169.254 friends), multicast, broadcast, the unspecified address,
13//!   IPv6 ULA/link-local, and IPv4-mapped IPv6. This is the primary defence against
14//!   SSRF — a malicious URL in a post must not let a client coerce our server into
15//!   probing its own internal network.
16//! - [`spawn_shutdown_signal_handler`] — wires SIGINT (Ctrl+C) and SIGTERM (or the
17//!   Windows console equivalents) into a `CancellationToken` so the server can shut
18//!   down gracefully; a second signal exits immediately.
19
20use log::warn;
21use tokio_util::sync::CancellationToken;
22
23pub async fn get_public_ipv4(force_local_network: bool) -> anyhow::Result<String> {
24    match force_local_network {
25        true => Ok("127.0.0.1".to_string()),
26        false => {
27            let client = reqwest::ClientBuilder::new().danger_accept_invalid_certs(true).build()?; // we are willing to accept invalid certs as we do application level signature checking
28            let response = client.get("https://api4.ipify.org").send().await?.text().await?;
29            Ok(response)
30        }
31    }
32}

34/// Cancels `cancellation_token` on the first shutdown request, and exits the process on a second one
35/// (for when graceful shutdown is stuck, e.g. during the identity PoW, which doesn't watch the token).
36///
37/// Unix: SIGINT (Ctrl+C) and SIGTERM (systemd, Docker, `kill`). Windows: Ctrl+C, Ctrl+Break, console close
38/// and system shutdown.
39///
40/// The OS handlers are registered before this returns, so a signal arriving straight after the call is
41/// never lost, and a SIGTERM never takes its default action of killing the process outright.
42pub fn spawn_shutdown_signal_handler(cancellation_token: CancellationToken) -> anyhow::Result<()> {
43    spawn_shutdown_signal_handler_with_force_exit(cancellation_token, || std::process::exit(130))
44}
45
46fn spawn_shutdown_signal_handler_with_force_exit(cancellation_token: CancellationToken, force_exit: impl FnOnce() + Send + 'static) -> anyhow::Result<()> {
47    let mut shutdown_signals = ShutdownSignals::register()?;
48    tokio::spawn(async move {
49        let signal_name = shutdown_signals.recv().await;
50        warn!("{} received, shutting down gracefully (send it again to exit immediately)...", signal_name);
51        cancellation_token.cancel();
52
53        let signal_name = shutdown_signals.recv().await;
54        warn!("{} received again, exiting immediately", signal_name);
55        force_exit();
56    });
57    Ok(())
58}

60#[cfg(unix)]
61struct ShutdownSignals {
62    sigint: tokio::signal::unix::Signal,
63    sigterm: tokio::signal::unix::Signal,
64}
65
66#[cfg(unix)]
67impl ShutdownSignals {
68    fn register() -> anyhow::Result<Self> {
69        use tokio::signal::unix::{signal, SignalKind};
70        Ok(Self { sigint: signal(SignalKind::interrupt())?, sigterm: signal(SignalKind::terminate())? })
71    }
72
73    async fn recv(&mut self) -> &'static str {
74        tokio::select! {
75            _ = self.sigint.recv() => "SIGINT",
76            _ = self.sigterm.recv() => "SIGTERM",
77        }
78    }
79}

81#[cfg(windows)]
82struct ShutdownSignals {
83    ctrl_c: tokio::signal::windows::CtrlC,
84    ctrl_break: tokio::signal::windows::CtrlBreak,
85    ctrl_close: tokio::signal::windows::CtrlClose,
86    ctrl_shutdown: tokio::signal::windows::CtrlShutdown,
87}
88
89#[cfg(windows)]
90impl ShutdownSignals {
91    fn register() -> anyhow::Result<Self> {
92        use tokio::signal::windows::{ctrl_break, ctrl_c, ctrl_close, ctrl_shutdown};
93        Ok(Self { ctrl_c: ctrl_c()?, ctrl_break: ctrl_break()?, ctrl_close: ctrl_close()?, ctrl_shutdown: ctrl_shutdown()? })
94    }
95
96    async fn recv(&mut self) -> &'static str {
97        tokio::select! {
98            _ = self.ctrl_c.recv() => "Ctrl+C",
99            _ = self.ctrl_break.recv() => "Ctrl+Break",
100            _ = self.ctrl_close.recv() => "Console close",
101            _ = self.ctrl_shutdown.recv() => "System shutdown",
102        }
103    }
104}

106/// Returns true for any IP that must never be the target of a server-initiated fetch.
107/// Covers: loopback, RFC-1918 private, link-local (incl. cloud metadata 169.254.169.254),
108/// CGNAT (100.64.0.0/10), multicast, unspecified, broadcast, IPv6 ULA (fc00::/7),
109/// IPv6 link-local (fe80::/10), and IPv4-mapped IPv6 addresses whose embedded v4 is protected.
110pub fn is_ssrf_protected_ip(ip: std::net::IpAddr) -> bool {
111    match ip {
112        std::net::IpAddr::V4(v4) => {
113            let o = v4.octets();
114            v4.is_loopback()                               // 127.0.0.0/8
115                || v4.is_private()                         // 10/8, 172.16/12, 192.168/16
116                || v4.is_link_local()                      // 169.254.0.0/16 — cloud metadata
117                || v4.is_multicast()                       // 224.0.0.0/4
118                || v4.is_unspecified()                     // 0.0.0.0
119                || v4.is_broadcast()                       // 255.255.255.255
120                || (o[0] == 100 && (o[1] & 0xC0) == 64)   // CGNAT 100.64.0.0/10
121        }
122        std::net::IpAddr::V6(v6) => {
123            let s = v6.segments();
124            v6.is_loopback()                               // ::1
125                || v6.is_multicast()                       // ff00::/8
126                || v6.is_unspecified()                     // ::
127                || (s[0] & 0xFFC0) == 0xFE80              // link-local fe80::/10
128                || (s[0] & 0xFE00) == 0xFC00              // unique local fc00::/7
129                // IPv4-mapped (::ffff:0:0/96) — recurse on the embedded v4 address
130                || matches!(v6.to_ipv4_mapped(), Some(v4) if is_ssrf_protected_ip(std::net::IpAddr::V4(v4)))
131        }
132    }
133}

135// Unix only: these signal this test process itself, which is safe under nextest (one process per test).
136#[cfg(all(test, unix))]
137mod tests {
138    use crate::tools::tools::spawn_shutdown_signal_handler_with_force_exit;
139    use std::time::Duration;
140    use tokio::sync::oneshot;
141    use tokio_util::sync::CancellationToken;
142
143    fn send_signal_to_self(signal_flag: &str) {
144        let status = std::process::Command::new("kill").args([signal_flag, &std::process::id().to_string()]).status().expect("failed to run kill");
145        assert!(status.success(), "kill {} failed", signal_flag);
146    }
147
148    fn spawn_handler_with_recorded_force_exit(cancellation_token: &CancellationToken) -> oneshot::Receiver<()> {
149        let (force_exit_sender, force_exit_receiver) = oneshot::channel::<()>();
150        spawn_shutdown_signal_handler_with_force_exit(cancellation_token.clone(), move || {
151            let _ = force_exit_sender.send(());
152        })
153        .expect("failed to register signal handlers");
154        force_exit_receiver
155    }
156
157    async fn assert_signal_cancels_token_without_force_exit(signal_flag: &str) {
158        let cancellation_token = CancellationToken::new();
159        let mut force_exit_receiver = spawn_handler_with_recorded_force_exit(&cancellation_token);
160
161        send_signal_to_self(signal_flag);
162
163        tokio::time::timeout(Duration::from_secs(5), cancellation_token.cancelled()).await.unwrap_or_else(|_| panic!("{} did not cancel the token", signal_flag));
164        assert!(tokio::time::timeout(Duration::from_millis(200), &mut force_exit_receiver).await.is_err(), "a single {} forced an exit", signal_flag);
165    }
166
167    #[tokio::test]
168    async fn sigterm_cancels_token() {
169        assert_signal_cancels_token_without_force_exit("-TERM").await;
170    }
171
172    #[tokio::test]
173    async fn sigint_cancels_token() {
174        assert_signal_cancels_token_without_force_exit("-INT").await;
175    }
176
177    #[tokio::test]
178    async fn second_signal_forces_exit() {
179        let cancellation_token = CancellationToken::new();
180        let force_exit_receiver = spawn_handler_with_recorded_force_exit(&cancellation_token);
181
182        send_signal_to_self("-TERM");
183        tokio::time::timeout(Duration::from_secs(5), cancellation_token.cancelled()).await.expect("SIGTERM did not cancel the token");
184
185        send_signal_to_self("-INT");
186        tokio::time::timeout(Duration::from_secs(5), force_exit_receiver).await.expect("second signal did not force an exit").expect("force_exit sender dropped");
187    }
188}
