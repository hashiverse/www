1//! # HTTPS-transport ownership proof
2//!
3//! Produces and verifies the proof bytes carried in an `AnnounceV2` from / to an
4//! HTTPS server. The bytes are a postcard-encoded `HttpsAcmeProof` containing
5//! the announcer's currently-serving ACME-issued TLS chain — nothing else.
6//!
7//! Verification is purely offline:
8//! 1. postcard decode (wrong-transport / corrupt bytes fail here),
9//! 2. [`hashiverse_lib::tools::cert_validation::is_cert_valid`] — X.509 path
10//!    validation against the bundled Mozilla NSS roots plus IP-SAN match.
11//!
12//! ### What V2 guarantees, and what it doesn't
13//!
14//! V2's only claim is "someone managed to issue a public-CA cert for the announced
15//! IP". That defeats the original target: a server that fails to acquire a cert
16//! (no control of port 80/443) can't construct a valid proof and never enters
17//! Kademlia. It does **not** defeat an attacker who borrows a stranger's chain —
18//! the chain is public, served on every TLS handshake. But any RPC the receiver
19//! then sends to the announced IP reaches the *real* server with the *real*
20//! identity at that IP, and the response-side identity mismatch trips the existing
21//! prune path (`HashiverseServer::add_potential_peer_to_kademlia` + RPC-failure
22//! pruning in `maintain_kademlia`). So the chain-borrowing attack is bounded: an
23//! attacker can briefly inject a wrong identity for an IP, and the network
24//! self-corrects on the next round-trip.
25

36/// Wire-encoded HTTPS ownership proof — the payload bytes inside an `AnnounceV2`
37/// when the announcer is an HTTPS-transport server. Serialised with postcard so
38/// other transports' verifiers fail to deserialise it and reject the announce,
39/// which is the cross-transport-mismatch path.
40#[derive(Serialize, Deserialize)]
41struct HttpsAcmeProof {
42    /// DER-encoded leaf first, then intermediate(s). Exactly what
43    /// `CertifiedKey::cert` carries server-side.
44    chain_der: Vec<Vec<u8>>,
45}
46
47/// Cached payload: the encoded proof for a given `Arc<CertifiedKey>` (compared by
48/// pointer equality — the cert refresher swaps in a new `Arc` on every reload).
49/// Reused on every announce tick until ACME rotates the cert, which is hours/days
50/// apart. The encoded bytes don't depend on the announcing peer (the chain is the
51/// only signed-by-nothing payload), so the cache key is just the cert pointer.
52struct CachedProofPayload {
53    certified_key: Arc<CertifiedKey>,
54    payload_bytes: Bytes,
55}
56
57/// Server-side proof object for the HTTPS transport. Holds one handle: a clone of
58/// `HttpsTransportCertRefresher::base_cert`, so every call snapshots the freshest
59/// loaded chain without coordinating with the cert refresher's own loop.
60///
61/// One instance per server, constructed in `FullHttpsTransportServer::new()` and
62/// shared via `Arc` from `get_transport_ownership_proof()` — so the cache survives
63/// across announce ticks.
64pub struct HttpsTransportOwnershipProof {
65    base_cert: Arc<RwLock<Option<Arc<CertifiedKey>>>>,
66    cached_payload: Mutex<Option<CachedProofPayload>>,
67}
68
69impl HttpsTransportOwnershipProof {
70    pub fn new(base_cert: Arc<RwLock<Option<Arc<CertifiedKey>>>>) -> Self {
71        Self { base_cert, cached_payload: Mutex::new(None) }
72    }
73}
74
75impl TransportOwnershipProof for HttpsTransportOwnershipProof {
76    fn make_ownership_proof_payload(&self) -> Option<Bytes> {
77        // Snapshot the current cert under the read lock and release it before doing
78        // anything expensive. The refresher writes a brand-new Arc<CertifiedKey> on each
79        // reload, so cache identity is settled by Arc::ptr_eq against this snapshot.
80        let current_certified_key: Arc<CertifiedKey> = {
81            let base_cert_guard = self.base_cert.read();
82            base_cert_guard.as_ref()?.clone()
83        };
84
85        // Fast path: cache still matches the current cert → reuse the encoded payload
86        // (a Bytes clone is just a pointer bump). 99%+ of calls hit this branch since
87        // announces fire every ~30s and ACME rotates the cert every few days.
88        {
89            let cache_guard = self.cached_payload.lock();
90            if let Some(cached) = cache_guard.as_ref() {
91                if Arc::ptr_eq(&cached.certified_key, &current_certified_key) {
92                    return Some(cached.payload_bytes.clone());
93                }
94            }
95        }
96
97        // Slow path: cert rotated (or first call). Re-encode.
98        let chain_der: Vec<Vec<u8>> = current_certified_key.cert.iter().map(|der| der.to_vec()).collect();
99        if chain_der.is_empty() {
100            return None;
101        }
102
103        let proof: HttpsAcmeProof = HttpsAcmeProof { chain_der };
104        let encoded: Vec<u8> = postcard::to_allocvec(&proof).ok()?;
105        let payload_bytes: Bytes = Bytes::from(encoded);
106
107        *self.cached_payload.lock() = Some(CachedProofPayload {
108            certified_key: current_certified_key,
109            payload_bytes: payload_bytes.clone(),
110        });
111
112        Some(payload_bytes)
113    }
114
115    fn prove(&self, peer: &Peer, proof_payload: &[u8], now: TimeMillis) -> bool {
116        let proof: HttpsAcmeProof = match postcard::from_bytes(proof_payload) {
117            Ok(p) => p,
118            Err(_) => return false,
119        };
120
121        is_cert_valid(&proof.chain_der, &peer.address, now)
122    }
123}

125#[cfg(test)]
126mod tests {
127    use super::*;
128    use hashiverse_lib::tools::types::Id;
129
130    fn peer_at(address: &str) -> Peer {
131        let mut peer: Peer = Peer::zero();
132        peer.address = address.to_string();
133        peer.id = Id([42u8; 32]);
134        peer
135    }
136
137    fn empty_base_cert_lock() -> Arc<RwLock<Option<Arc<CertifiedKey>>>> {
138        Arc::new(RwLock::new(None))
139    }
140
141    #[test]
142    fn make_returns_none_when_no_cert_loaded_yet() {
143        let proof: HttpsTransportOwnershipProof = HttpsTransportOwnershipProof::new(empty_base_cert_lock());
144        assert!(proof.make_ownership_proof_payload().is_none());
145    }
146
147    #[test]
148    fn prove_rejects_garbage_bytes() {
149        let proof: HttpsTransportOwnershipProof = HttpsTransportOwnershipProof::new(empty_base_cert_lock());
150        let peer: Peer = peer_at("1.2.3.4:443");
151        let now: TimeMillis = TimeMillis(1_700_000_000_000);
152        assert!(!proof.prove(&peer, &[0xff, 0xfe, 0xfd], now));
153    }
154
155    #[test]
156    fn prove_rejects_empty_bytes() {
157        let proof: HttpsTransportOwnershipProof = HttpsTransportOwnershipProof::new(empty_base_cert_lock());
158        let peer: Peer = peer_at("1.2.3.4:443");
159        let now: TimeMillis = TimeMillis(1_700_000_000_000);
160        // Empty bytes = the mem-transport marker. The HTTPS verifier must reject it
161        // (cross-transport mismatch): postcard either decodes an empty slice into
162        // HttpsAcmeProof { chain_der: vec![] }, in which case is_cert_valid then fails
163        // on the empty chain — or postcard rejects the bytes outright. Either way: false.
164        assert!(!proof.prove(&peer, &[], now));
165    }
166
167    /// Build a `CertifiedKey` carrying placeholder DER bytes. The payload won't survive
168    /// real X.509 verification (that's `is_cert_valid`'s job, exercised separately), but
169    /// it's sufficient to drive cache-hit / cache-miss bookkeeping in the proof object.
170    fn placeholder_certified_key(der_bytes: Vec<u8>) -> Arc<CertifiedKey> {
171        use rustls_pki_types::{CertificateDer, PrivateKeyDer, PrivatePkcs8KeyDer};
172        let _ = rustls::crypto::aws_lc_rs::default_provider().install_default();
173        let cert_chain: Vec<CertificateDer<'static>> = vec![CertificateDer::from(der_bytes)];
174        // rcgen ships a key generator; we only need a structurally-valid PKCS#8 envelope
175        // so rustls' any_supported_type accepts it. The cert above is bogus DER so the
176        // CertifiedKey can never sign anything real — fine, the cache tests don't sign.
177        let key_pair: rcgen::KeyPair = rcgen::KeyPair::generate().expect("rcgen key gen never fails on supported algorithms");
178        let pkcs8: Vec<u8> = key_pair.serialize_der();
179        let key_der: PrivateKeyDer<'static> = PrivateKeyDer::Pkcs8(PrivatePkcs8KeyDer::from(pkcs8));
180        let signing: Arc<dyn rustls::sign::SigningKey> = rustls::crypto::aws_lc_rs::sign::any_supported_type(&key_der).expect("rcgen key is a supported type");
181        Arc::new(CertifiedKey { cert: cert_chain, key: signing, ocsp: None })
182    }
183
184    #[test]
185    fn cache_hit_returns_identical_bytes_when_cert_unchanged() {
186        let base_cert: Arc<RwLock<Option<Arc<CertifiedKey>>>> = Arc::new(RwLock::new(Some(placeholder_certified_key(vec![0x30, 0x82, 0x01, 0x00, 0xAA]))));
187        let proof: HttpsTransportOwnershipProof = HttpsTransportOwnershipProof::new(base_cert);
188
189        let first_bytes: Bytes = proof.make_ownership_proof_payload().expect("placeholder cert is non-empty");
190        let second_bytes: Bytes = proof.make_ownership_proof_payload().expect("placeholder cert is non-empty");
191
192        // Pointer equality on the underlying buffer: the cache returned the cached `Bytes`,
193        // not a freshly-encoded one.
194        assert_eq!(first_bytes.as_ptr(), second_bytes.as_ptr());
195        assert_eq!(first_bytes, second_bytes);
196    }
197
198    #[test]
199    fn cache_miss_when_cert_rotates() {
200        let base_cert: Arc<RwLock<Option<Arc<CertifiedKey>>>> = Arc::new(RwLock::new(Some(placeholder_certified_key(vec![0x01, 0x02, 0x03]))));
201        let proof: HttpsTransportOwnershipProof = HttpsTransportOwnershipProof::new(base_cert.clone());
202
203        let first_bytes: Bytes = proof.make_ownership_proof_payload().expect("placeholder cert is non-empty");
204
205        // Simulate ACME rotation: refresher writes a new Arc<CertifiedKey> into the lock.
206        *base_cert.write() = Some(placeholder_certified_key(vec![0x04, 0x05, 0x06]));
207
208        let second_bytes: Bytes = proof.make_ownership_proof_payload().expect("placeholder cert is non-empty");
209
210        assert_ne!(first_bytes.as_ptr(), second_bytes.as_ptr());
211        assert_ne!(first_bytes, second_bytes);
212    }
