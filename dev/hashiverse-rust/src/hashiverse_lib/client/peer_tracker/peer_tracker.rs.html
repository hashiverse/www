1//! # Kademlia-style local peer set
2//!
3//! `PeerTracker` is the client's working memory of the network: a deduplicated list of
4//! [`crate::protocol::peer::Peer`] records loaded from `BUCKET_PEER`, validated, and
5//! indexed by XOR distance so [`crate::client::peer_tracker::peer_iterator::PeerIterator`]
6//! can walk them in order of closeness to any target
7//! [`crate::tools::types::Id`].
8//!
9//! Responsibilities:
10//!
11//! - **Bootstrap** — when the tracker is empty (new install, wiped storage) it calls
12//!   `BootstrapV1` against seed domains from [`crate::tools::config::BOOTSTRAP_DOMAINS`]
13//!   to obtain a starting set of peers.
14//! - **Freshness** — stale peers (missed announces, failed RPCs) are demoted or dropped;
15//!   fresh ones from gossip (`AnnounceV2`) or from RPC responses are folded in.
16//! - **Flush** — peer mutations set `peers_need_flush`; the outer client batches those
17//!   into periodic writes to storage so every RPC response doesn't trigger a disk write.
18//!
19//! ## Locking
20//!
21//! The tracker is shared as a [`SharedPeerTracker`] (`Arc<parking_lot::RwLock<PeerTracker>>`) so that
22//! concurrent DHT walks can interleave.  Every critical section is synchronous and short: anything that
23//! awaits (RPCs, storage writes) or is expensive (peer verification) happens with no guard held.
24//! `parking_lot` guards are `!Send`, so holding one across an `.await` in a `Send` future fails to compile.
25

41/// The client-wide handle to the peer tracker.  Lock it only for short synchronous sections - never across an `.await`.
42pub type SharedPeerTracker = Arc<parking_lot::RwLock<PeerTracker>>;
43
44/// The client's local view of the known peer set and the entry point for Kademlia-style
45/// routing.
46///
47/// `PeerTracker` owns the in-memory list of [`Peer`] records the client has seen, persists
48/// them to [`ClientStorage`] under `BUCKET_PEER` so they survive restarts, and exposes the
49/// iteration primitives used throughout the client when it needs to answer "who should I
50/// talk to next about this [`Id`]?". When the list is empty (first launch, or after a
51/// reset), it seeds itself via a `BootstrapV1` RPC against the
52/// [`crate::transport::bootstrap_provider::bootstrap_provider::BootstrapProvider`] addresses configured on the
53/// transport.
54///
55/// The tracker is the single source of truth for peer freshness: stale or bad peers get
56/// evicted here, new peers get folded in here, and the `peers_need_flush` flag coalesces
57/// rapid updates into a single disk write.
58pub struct PeerTracker {
59    runtime_services: Arc<RuntimeServices>,
60    client_storage: Arc<dyn ClientStorage>,
61    peers_need_flush: bool,
62    peers: Vec<Peer>,
63    max_peers: usize,
64    /// Serialises flushes so a slow, older snapshot can never land in storage after a newer one.
65    flush_serialiser: Arc<tokio::sync::Mutex<()>>,
66}
67
68impl PeerTracker {
69    pub async fn new(runtime_services: Arc<RuntimeServices>, client_storage: Arc<dyn ClientStorage>) -> anyhow::Result<Self> {
70        Self::new_with_max_peers(runtime_services, client_storage, config::CLIENT_PEER_TRACKER_MAX_PEERS).await
71    }

73    pub async fn new_with_max_peers(runtime_services: Arc<RuntimeServices>, client_storage: Arc<dyn ClientStorage>, max_peers: usize) -> anyhow::Result<Self> {
74        let peers: anyhow::Result<Vec<Peer>> = try {
75            let peers = client_storage::get_struct::<Vec<Peer>>(client_storage.as_ref(), BUCKET_PEER, "peers", runtime_services.time_provider.current_time_millis()).await?;
76            match peers {
77                Some(peers) => {
78                    info!("PeerTracker is starting with {} peers", peers.len());
79                    trace!("{:?}", peers);
80                    peers
81                }
82                None => Vec::new(),
83            }
84        };
85
86        let peers = peers.unwrap_or_else(|e| {
87            warn!("Failed to load peers from storage: {}", e);
88            Vec::new()
89        });
90
91        let current_time_millis = runtime_services.time_provider.current_time_millis();
92        let (peers, peers_were_dropped) = Self::sanitise_loaded_peers(peers, current_time_millis, max_peers);
93
94        Ok(Self {
95            runtime_services,
96            client_storage,
97            peers_need_flush: peers_were_dropped,
98            peers,
99            max_peers,
100            flush_serialiser: Arc::new(tokio::sync::Mutex::new(())),
101        })
102    }

104    /// Persisted peers predate the current bounds: drop far-future records (which could otherwise never be replaced,
105    /// because newest-timestamp wins) and trim to `max_peers` by score.  Returns the peers sorted by id.
106    fn sanitise_loaded_peers(peers: Vec<Peer>, current_time_millis: TimeMillis, max_peers: usize) -> (Vec<Peer>, bool) {
107        let loaded_peer_count = peers.len();
108        let mut peers: Vec<Peer> = peers
109            .into_iter()
110            .filter(|peer| match peer.verify_timestamps(current_time_millis) {
111                Ok(()) => true,
112                Err(e) => {
113                    warn!("dropping stored peer: {}", e);
114                    false
115                }
116            })
117            .collect();
118
119        if peers.len() > max_peers {
120            peers.sort_by(|a, b| b.score(current_time_millis).total_cmp(&a.score(current_time_millis)));
121            peers.truncate(max_peers);
122        }
123        peers.sort_by_key(|peer| peer.id);
124
125        let peers_were_dropped = peers.len() != loaded_peer_count;
126        (peers, peers_were_dropped)
127    }

129    /// Clears the dirty flag and returns the peers to persist, or `None` if nothing changed since the last flush.
130    pub fn take_flush_snapshot(&mut self) -> Option<Vec<Peer>> {
131        if !self.peers_need_flush {
132            return None;
133        }
134        self.peers_need_flush = false;
135        Some(self.peers.clone())
136    }

146    /// Verify-and-insert in one go.  Holders of a [`SharedPeerTracker`] should instead call [`Self::needs_verification`] under a read lock,
147    /// [`Self::verify_candidate`] with no lock held, then [`Self::insert_verified`] under a write lock - verification is the expensive part.
148    pub fn add_peer(&mut self, peer: Peer) -> anyhow::Result<()> {
149        if !self.needs_verification(&peer) {
150            return Ok(());
151        }
152        Self::verify_candidate(&peer, self.current_time_millis())?;
153        self.insert_verified(peer);
154        Ok(())
155    }

157    /// Cheap dedupe before the expensive verify: servers resend the same peers in every peers_nearer list.
158    /// False if we already hold this peer with a timestamp at least as new.
159    pub fn needs_verification(&self, peer: &Peer) -> bool {
160        match self.peers.binary_search_by_key(&peer.id, |peer| peer.id) {
161            Ok(i) => peer.timestamp > self.peers[i].timestamp,
162            Err(_) => true,
163        }
164    }
165
166    /// Everything a peer must pass before it may enter the tracker.  Pure, so it can run with no lock held.
167    pub fn verify_candidate(peer: &Peer, current_time_millis: TimeMillis) -> anyhow::Result<()> {
168        // Sanity check that this peer is kosher
169        if let Err(e) = peer.verify() {
170            anyhow::bail!("peer verification error: {}", e);
171        }
172
173        // Check that its pow is reasonable
174        if peer.pow_initial.pow < config::SERVER_KEY_POW_MIN {
175            anyhow::bail!("peer peer.pow_initial.pow={} < {}", peer.pow_initial.pow, config::SERVER_KEY_POW_MIN);
176        }
177
178        // A far-future timestamp would win every later newest-timestamp comparison, so the real peer could never replace it
179        peer.verify_timestamps(current_time_millis)?;
180
181        Ok(())
182    }
183
184    /// Inserts a peer that has passed [`Self::verify_candidate`].  Re-checks the timestamp dedupe, because another walk may have
185    /// inserted a newer copy between our verify and this insert.
186    pub fn insert_verified(&mut self, peer: Peer) {
187        match self.peers.binary_search_by_key(&peer.id, |peer| peer.id) {
188            Ok(i) => {
189                if peer.timestamp <= self.peers[i].timestamp {
190                    return;
191                }
192                self.peers[i] = peer;
193            }
194            Err(i) => {
195                if self.peers.len() < self.max_peers {
196                    self.peers.insert(i, peer);
197                }
198                else {
199                    // Full: the newcomer only gets in by out-scoring our weakest peer
200                    let current_time_millis = self.current_time_millis();
201                    let weakest_peer = self.peers.iter().enumerate().map(|(index, existing_peer)| (index, existing_peer.score(current_time_millis))).min_by(|a, b| a.1.total_cmp(&b.1));
202                    let Some((weakest_peer_index, weakest_peer_score)) = weakest_peer else { return };
203                    if peer.score(current_time_millis) <= weakest_peer_score {
204                        return;
205                    }
206                    self.peers.remove(weakest_peer_index);
207                    let insert_index = self.peers.binary_search_by_key(&peer.id, |peer| peer.id).unwrap_or_else(|insert_index| insert_index);
208                    self.peers.insert(insert_index, peer);
209                }
210            }
211        }
212
213        self.peers_need_flush = true;
214    }
215
216    pub fn remove_peer(&mut self, peer: &Peer) {
217        if let Ok(i) = self.peers.binary_search_by_key(&peer.id, |peer| peer.id) {
218            self.peers.remove(i);
219            self.peers_need_flush = true;
220        }
221    }

237    /// Persists the peers if they changed.  The storage write happens with no tracker lock held.
238    pub async fn flush(peer_tracker: &SharedPeerTracker) -> anyhow::Result<()> {
239        let flush_serialiser = peer_tracker.read().flush_serialiser.clone();
240        let _flush_guard = flush_serialiser.lock().await;
241
242        let (client_storage, current_time_millis, peers) = {
243            let mut peer_tracker = peer_tracker.write();
244            let Some(peers) = peer_tracker.take_flush_snapshot() else { return Ok(()) };
245            (peer_tracker.client_storage.clone(), peer_tracker.current_time_millis(), peers)
246        };
247
248        trace!("Flushing peers to storage");
249        let result = client_storage::put_struct(client_storage.as_ref(), BUCKET_PEER, "peers", &peers, current_time_millis).await;
250        if result.is_err() {
251            peer_tracker.write().mark_needs_flush();
252        }
253        result
254    }

256    /// Bootstraps if necessary, then starts a walk towards `bucket_location_id`.  The iterator locks the tracker only briefly per step.
257    pub async fn iterate_to_location(peer_tracker: &SharedPeerTracker, bucket_location_id: Id, max_iterations_since_high_watermark: usize, cache_radius: Option<LeadingAgreementBits>) -> anyhow::Result<PeerIterator> {
258        Self::bootstrap(peer_tracker).await?;
259
260        Ok(PeerIterator::new(peer_tracker.clone(), bucket_location_id, max_iterations_since_high_watermark, cache_radius))
261    }

263    /// Seeds an empty tracker from the bootstrap addresses.  RPCs and verification happen with no lock held; two concurrent
264    /// bootstraps may both fetch, which is harmless because inserts are idempotent.
265    pub async fn bootstrap(peer_tracker: &SharedPeerTracker) -> anyhow::Result<()> {
266        // We only need to bootstrap if we have noone to talk to!
267        let runtime_services = {
268            let peer_tracker = peer_tracker.read();
269            if !peer_tracker.is_empty() {
270                return Ok(());
271            }
272            peer_tracker.runtime_services.clone()
273        };
274
275        info!("bootstrapping PeerTracker");
276
277        // Lets randomize these addresses so that the first one is not snowed
278        let mut bootstrap_addresses = runtime_services.transport_factory.get_bootstrap_addresses().await;
279        tools::shuffle(&mut bootstrap_addresses);
280
281        // Our bootstrap process has handed us a bunch of raw addresses, so we need to convert them into peers
282        for bootstrap_address in bootstrap_addresses {
283            let try_result: anyhow::Result<()> = try {
284                info!("bootstrapping {}", bootstrap_address);
285
286                let request = json::struct_to_bytes(&BootstrapV1 {})?;
287                let response = rpc::rpc::rpc_server_unknown(&runtime_services, &Id::zero(), &bootstrap_address, PayloadRequestKind::BootstrapV1, request).await?;
288                anyhow_assert_eq!(&PayloadResponseKind::BootstrapResponseV1, &response.response_request_kind);
289                let response = json::bytes_to_struct::<BootstrapResponseV1>(&response.bytes)?;
290                if response.peers_random.len() > config::CLIENT_MAX_PEERS_CONSIDERED_PER_RESPONSE {
291                    warn!("ignoring {} of {} bootstrapped peers from {} (cap is {})", response.peers_random.len() - config::CLIENT_MAX_PEERS_CONSIDERED_PER_RESPONSE, response.peers_random.len(), bootstrap_address, config::CLIENT_MAX_PEERS_CONSIDERED_PER_RESPONSE);
292                }
293
294                let current_time_millis = runtime_services.time_provider.current_time_millis();
295                let verified_peers: Vec<Peer> = response
296                    .peers_random
297                    .into_iter()
298                    .take(config::CLIENT_MAX_PEERS_CONSIDERED_PER_RESPONSE)
299                    .filter(|peer| match Self::verify_candidate(peer, current_time_millis) {
300                        Ok(()) => true,
301                        Err(e) => {
302                            warn!("problem while adding bootstrapped peer: {}", e);
303                            false
304                        }
305                    })
306                    .collect();
307
308                let mut peer_tracker = peer_tracker.write();
309                for peer in verified_peers {
310                    peer_tracker.insert_verified(peer);
311                }
312            };
313
314            if let Err(e) = try_result {
315                warn!("problem bootstrapping peer {}: {}", bootstrap_address, e);
316            }
317
318            // We only need to continue bootstrapping if we still have noone to talk to!
319            if !peer_tracker.read().is_empty() {
320                break;
321            }
322        }
323
324        Ok(())
325    }

481        #[tokio::test]
482        async fn full_tracker_evicts_weakest_peer_only_for_a_stronger_newcomer() -> anyhow::Result<()> {
483            let (runtime_services, manual_time_provider) = manual_runtime_services();
484            let client_storage = MemClientStorage::new().await?;
485            let mut peer_tracker = PeerTracker::new_with_max_peers(runtime_services.clone(), client_storage, 2).await?;
486
487            // Two peers whose pows are 90 days old by the time the newcomer arrives: their score has decayed away.
488            let (_, stale_peer_1) = make_peer(&runtime_services).await?;
489            let (_, stale_peer_2) = make_peer(&runtime_services).await?;
490            // A peer that will be fresh at the end, created after the clock moves on.
491            manual_time_provider.set_time(START_TIME_MILLIS + MILLIS_IN_DAY.const_mul(90));
492            let (_, fresh_peer) = make_peer(&runtime_services).await?;
493            let (_, other_fresh_peer) = make_peer(&runtime_services).await?;
494            let now = runtime_services.time_provider.current_time_millis();
495            let weakest_fresh_score = fresh_peer.score(now).min(other_fresh_peer.score(now));
496            let strongest_stale_score = stale_peer_1.score(now).max(stale_peer_2.score(now));
497            assert!(weakest_fresh_score > strongest_stale_score, "test setup: fresh peers must out-score stale peers");
498
499            // Full with {stale, fresh}: a stronger newcomer evicts the stale peer and keeps the fresh one.
500            peer_tracker.add_peer(stale_peer_1.clone())?;
501            peer_tracker.add_peer(fresh_peer.clone())?;
502            peer_tracker.add_peer(other_fresh_peer.clone())?;
503            let mut expected_ids = vec![fresh_peer.id, other_fresh_peer.id];
504            expected_ids.sort();
505            assert_eq!(expected_ids, peer_ids(&peer_tracker), "the weakest peer must be evicted for a stronger newcomer");
