1//! # `EncodedPostBundleFeedbackV1` — server-signed aggregate feedback for a bundle
2//!
3//! Feedback (reactions, upvotes/downvotes, flags) on the posts in a single
4//! `(location, time bucket)` is aggregated into its own bundle, parallel to — but
5//! independent of — the [`crate::protocol::posting::encoded_post_bundle`]. The two are
6//! split because feedback can change long after the post bundle itself has sealed, and
7//! many consumers don't need the feedback layer at all.
8//!
9//! Wire format:
10//!
11//! - **Header** (server-signed) — lists every feedback entry's `(post_id,
12//!   feedback_type, salt, pow)` along with a hash of the body. Signed by the
13//!   publishing [`crate::protocol::peer::Peer`].
14//! - **Body** — a packed array of fixed-size
15//!   [`crate::protocol::posting::encoded_post_feedback::EncodedPostFeedbackV1`] entries,
16//!   exactly `N * ENTRY_SIZE` bytes. Fixed sizing means iteration and verification are
17//!   O(N) with no per-entry header overhead.
18//!
19//! `verify()` checks the signature, confirms body length is a multiple of `ENTRY_SIZE`,
20//! re-hashes the body and compares against the header's hash, and checks the
21//! proof-of-work on every entry against the numeraire — Sybil-resistant voting in one
22//! pass.
23

35// Contains all the posts for a given location_id in a particular bucket
36#[derive(Serialize, Deserialize, Debug, PartialEq, Clone)]
37pub struct EncodedPostBundleFeedbackHeaderV1 {
38    pub time_millis: TimeMillis, // When was this bundle last updated
39    pub location_id: Id,
40    pub feedbacks_bytes_hash: Hash, // Hash(feedbacks_bytes from the body)
41    pub peer: Peer,
42
43    pub signature: Signature, // Peer.sign(get_hash_for_signing())
44}
45
46impl EncodedPostBundleFeedbackHeaderV1 {
47    pub fn get_hash_for_signing(&self) -> Hash {
48        let time_millis_be = self.time_millis.encode_be();
49
50        let hash_input: Vec<&[u8]> = vec![
51            time_millis_be.as_ref(),
52            self.location_id.as_ref(),
53            self.feedbacks_bytes_hash.as_ref(),
54            self.peer.signature.as_ref(),
55        ];
56
57        hashing::hash_multiple(&hash_input)
58    }
59
60    pub fn signature_generate(&mut self, signature_key: &SignatureKey) {
61        let hash = self.get_hash_for_signing();
62        self.signature = signing::sign(signature_key, hash.as_ref());
63    }
64
65    pub fn signature_verify(&self) -> anyhow::Result<()> {
66        let hash = self.get_hash_for_signing();
67        let verification_key = VerificationKey::from_bytes(&self.peer.verification_key_bytes)?;
68        signing::verify(&verification_key, &self.signature, hash.as_ref())
69    }
70
71    pub fn verify(&self) -> anyhow::Result<()> {
72        self.signature_verify()?;
73        Ok(())
74    }
75}

94impl EncodedPostBundleFeedbackV1 {
95    pub fn to_bytes(&self) -> anyhow::Result<Bytes> {
96        let mut bytes = BytesMut::new();
97
98        let header = json::struct_to_bytes(&self.header)?;
99        bytes.put_u8(1u8); // Version
100        bytes.put_u64(header.len() as u64);
101        bytes.put_u64(self.feedbacks_bytes.len() as u64);
102        bytes.put_slice(header.as_ref());
103        bytes.put_slice(self.feedbacks_bytes.as_ref());
104
105        Ok(bytes.freeze())
106    }
107
108    pub fn from_bytes(mut bytes: Bytes) -> anyhow::Result<Self> {
109        if bytes.remaining() < 1 {
110            anyhow::bail!("Invalid buffer: missing post_bundle version");
111        }
112        let version = bytes.get_u8();
113        if 1 != version {
114            anyhow::bail!("Invalid buffer: unknown post_bundle version");
115        }
116
117        if bytes.remaining() < 16 {
118            anyhow::bail!("Invalid buffer: missing post_bundle lengths");
119        }
120
121        let header_len = bytes.get_u64() as usize;
122        let body_len = bytes.get_u64() as usize;
123        let total_length = header_len.checked_add(body_len).ok_or_else(|| anyhow::anyhow!("total_length overflow"))?;
124        if bytes.remaining() < total_length {
125            anyhow::bail!("Invalid buffer: post_bundle data truncated");
126        }
127        let header_bytes = bytes.copy_to_bytes(header_len);
128        let body_bytes = bytes.copy_to_bytes(body_len);
129
130        let header = json::bytes_to_struct(&header_bytes)?;
131
132        Ok(EncodedPostBundleFeedbackV1 { header, feedbacks_bytes: body_bytes })
133    }
134
135    /// Verifies that this bundle is legitimate enough to cache.
136    ///
137    /// Checks:
138    /// 1. The bundle header is valid and its signature verifies.
139    /// 2. The feedback body is a valid multiple of `ENTRY_SIZE` (no partial entries).
140    /// 3. The header's hash is the same as the body's
141    /// 4. Each feedback entry's proof-of-work is correct.
142    pub fn verify(&self) -> anyhow::Result<()> {
143        self.verify_envelope()?;
144        self.verify_entries_pow()
145    }
146
147    /// Checks (1)-(3) of [`Self::verify`]: the bundle is intact and signed as its header claims, without checking each entry's PoW.
148    fn verify_envelope(&self) -> anyhow::Result<()> {
149        // (1) Header signature
150        self.header.verify()?;
151
152        // (2) Body must be an exact multiple of the fixed entry size
153        if !self.feedbacks_bytes.len().is_multiple_of(ENTRY_SIZE) {
154            anyhow::bail!(
155                "feedbacks_bytes length ({}) is not a multiple of ENTRY_SIZE ({})",
156                self.feedbacks_bytes.len(),
157                ENTRY_SIZE
158            );
159        }
160
161        // (3) Hash
162        let feedbacks_bytes_hash = hashing::hash(self.feedbacks_bytes.as_ref());
163        anyhow_assert_eq!(feedbacks_bytes_hash, self.header.feedbacks_bytes_hash, "feedbacks_bytes_hash mismatch");
164        Ok(())
165    }
166
167    /// Check (4) of [`Self::verify`].
168    fn verify_entries_pow(&self) -> anyhow::Result<()> {
169        // (4) PoW check for every entry
170        let num_entries = self.feedbacks_bytes.len() / ENTRY_SIZE;
171        for i in 0..num_entries {
172            let entry_bytes = &self.feedbacks_bytes[i * ENTRY_SIZE..(i + 1) * ENTRY_SIZE];
173            let feedback = EncodedPostFeedbackV1::decode_from_bytes(&mut &entry_bytes[..])
174                .map_err(|e| anyhow::anyhow!("feedback {}: invalid entry: {}", i, e))?;
175            feedback.pow_verify()
176                .map_err(|e| anyhow::anyhow!("feedback {}: pow verification failed: {}", i, e))?;
177        }
178
179        Ok(())
180    }
181
182    /// Verifies that this bundle is a legitimate feedback bundle for `location_id`, as received from the network.
183    ///
184    /// On top of [`EncodedPostBundleFeedbackV1::verify`]:
185    /// 1. The publishing peer is a genuine server identity (server-id PoW, id derived from its key, self-signed).
186    ///    [`EncodedPostBundleFeedbackHeaderV1::verify`] alone only checks the header against the key the header itself supplies.
187    /// 2. The bundle is for this location_id.
188    pub fn verify_for_location(&self, location_id: &Id) -> anyhow::Result<()> {
189        self.verify_origin_for_location(location_id)?;
190        self.verify_entries_pow()
191    }
192
193    /// [`Self::verify_for_location`] minus the per-entry PoW check: proves a genuine server published this bundle for `location_id`,
194    /// but tolerates individual entries with bad PoW, which [`Self::merge`] drops one at a time.  For a server's own (direct) copy, so that
195    /// one bad entry it holds cannot erase all of its other feedback.  Never forward a bundle that has only passed this check.
196    pub fn verify_origin_for_location(&self, location_id: &Id) -> anyhow::Result<()> {
197        self.header.peer.verify().map_err(|e| anyhow::anyhow!("feedback bundle peer {} failed verification: {}", self.header.peer.id, e))?;
198        if self.header.location_id != *location_id {
199            anyhow::bail!("feedback bundle location_id {} does not match location_id {}", self.header.location_id, location_id);
200        }
201        self.verify_envelope()
202    }
203
204    pub fn get_post_pow_for_feedback_type(&self, post_id: &Id, feedback_type: u8) -> Pow {
205        for view in EncodedPostFeedbackViewV1::iter(&self.feedbacks_bytes).flatten() {
206            // Check feedback_type first as it short circuits and is therefore faster...
207            if view.feedback_type() == feedback_type && view.post_id_bytes() == post_id.as_ref() {
208                return view.pow();
209            }
210        }
211        Pow(0)
212    }
213
214    /// Unions an array of bundles by taking the highest-pow entry for each
215    /// `(post_id, feedback_type)` pair across all bundles.  The header is
216    /// taken from the most-recently-timestamped bundle.  Returns `None` if
217    /// `bundles` is empty.
218    ///
219    /// Entries whose claimed pow does not match what their salt achieves are
220    /// dropped individually.  The bundle header signature only proves which
221    /// server produced the bundle, not that its entries are honest, so a
222    /// dishonest (or poisoned) server must not be able to plant pow=255 flags.
223    /// Dropping per-entry rather than rejecting the whole bundle means one bad
224    /// entry cannot erase all other feedback for the bucket.
225    pub fn merge(bundles: &[Self]) -> Option<Self> {
226        let header = bundles.iter().max_by_key(|b| b.header.time_millis)?.header.clone();
227
228        let mut global_max: HashMap<(Id, u8), EncodedPostFeedbackV1> = HashMap::new();
229        for bundle in bundles {
230            for view in EncodedPostFeedbackViewV1::iter(&bundle.feedbacks_bytes) {
231                let Ok(view) = view else { continue };
232                let Ok(post_id) = Id::from_slice(view.post_id_bytes()) else { continue };
233                let Ok(salt) = Salt::from_slice(view.salt_bytes()) else { continue };
234                let encoded_post_feedback = EncodedPostFeedbackV1::new(post_id, view.feedback_type(), salt, view.pow());
235                if encoded_post_feedback.pow_verify().is_err() {
236                    continue;
237                }
238
239                let key = (post_id, encoded_post_feedback.feedback_type);
240                match global_max.entry(key) {
241                    std::collections::hash_map::Entry::Vacant(vacant) => {
242                        vacant.insert(encoded_post_feedback);
243                    }
244                    std::collections::hash_map::Entry::Occupied(mut occupied) => {
245                        if encoded_post_feedback.pow > occupied.get().pow {
246                            occupied.insert(encoded_post_feedback);
247                        }
248                    }
249                }
250            }
251        }
252
253        let mut feedbacks_bytes_mut = Vec::new();
254        for feedback in global_max.values() {
255            let _ = feedback.append_encode_to_bytes(&mut feedbacks_bytes_mut);
256        }
257
258        Some(Self { header, feedbacks_bytes: Bytes::from(feedbacks_bytes_mut) })
259    }
260
261    pub fn get_post_pows(&self, post_id: &Id) -> [Pow; 256] {
262        let mut result = [Pow(0); 256];
263        for view in EncodedPostFeedbackViewV1::iter(&self.feedbacks_bytes).flatten() {
264            if  view.post_id_bytes() == post_id.as_ref() {
265                result[view.feedback_type() as usize] = view.pow();
266            }
267        }
268
269        result
270    }
271}

491        let poisoned_bundle = make_bundle_with_encoded_post_feedbacks(TimeMillis(1000), &[make_forged_feedback(post_a, 1), valid_for_post_a.clone(), make_forged_feedback(post_b, 2)]).await?;
492        let honest_bundle = make_bundle_with_encoded_post_feedbacks(TimeMillis(2000), &[]).await?;

494        let merged = EncodedPostBundleFeedbackV1::merge(&[poisoned_bundle, honest_bundle]).expect("non-empty input");

496        assert_eq!(valid_for_post_a.pow, merged.get_post_pow_for_feedback_type(&post_a, 1), "the valid entry must survive alongside a forged one");
497        assert_eq!(Pow(0), merged.get_post_pow_for_feedback_type(&post_b, 2), "a key with only a forged entry must be absent");
498        assert_eq!(merged.feedbacks_bytes.len(), ENTRY_SIZE, "only the one valid entry should remain");

500        Ok(())

503    // merge() output is a client-side view: its header (taken from the newest bundle) no longer covers its
504    // feedbacks_bytes, so it must never be forwarded to caches - only intact originals that pass verify() may be.
505    #[tokio::test]
506    async fn merged_bundle_is_a_client_view_not_a_signed_bundle() -> anyhow::Result<()> {
507        let post_a = Id::random();
508        let valid_for_type_1 = make_valid_feedback(post_a, 1)?;
509        let valid_for_type_2 = make_valid_feedback(post_a, 2)?;
510
511        let bundle_first = make_bundle_with_encoded_post_feedbacks(TimeMillis(1000), &[valid_for_type_1.clone()]).await?;
512        let bundle_second = make_bundle_with_encoded_post_feedbacks(TimeMillis(2000), &[valid_for_type_2.clone()]).await?;
513        bundle_first.verify()?;
514        bundle_second.verify()?;
515
516        let merged = EncodedPostBundleFeedbackV1::merge(&[bundle_first, bundle_second]).expect("non-empty input");
517        assert_eq!(valid_for_type_1.pow, merged.get_post_pow_for_feedback_type(&post_a, 1));
518        assert_eq!(valid_for_type_2.pow, merged.get_post_pow_for_feedback_type(&post_a, 2));
519        assert!(merged.verify().is_err(), "the merged union is not covered by any originator's signature");
520
521        // A signed bundle carrying a forged entry is not uploadable, yet merge still salvages its honest entry for the client view.
522        let poisoned_bundle = make_bundle_with_encoded_post_feedbacks(TimeMillis(3000), &[make_forged_feedback(post_a, 3), valid_for_type_1.clone()]).await?;
523        poisoned_bundle.header.verify()?;
524        assert!(poisoned_bundle.verify().is_err(), "a bundle with a forged entry must fail full verification");
525        let merged_from_poisoned = EncodedPostBundleFeedbackV1::merge(&[poisoned_bundle]).expect("non-empty input");
526        assert_eq!(valid_for_type_1.pow, merged_from_poisoned.get_post_pow_for_feedback_type(&post_a, 1));
527        assert_eq!(Pow(0), merged_from_poisoned.get_post_pow_for_feedback_type(&post_a, 3));
528        Ok(())
529    }
