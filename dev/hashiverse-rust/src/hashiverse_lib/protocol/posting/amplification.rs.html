1//! # Post PoW amplification rules
2//!
3//! A single function — [`get_minimum_post_pow`] — that decides how many bits of
4//! proof-of-work a post requires before a server will accept it. The base level comes
5//! from [`crate::tools::config::POW_MINIMUM_PER_POST`], on top of which the
6//! amplification is:
7//!
8//! - **Size**: log-scale above 1024 bytes. Longer posts cost more — fair for storage,
9//!   punishes spam-generated wall-of-text content.
10//! - **Fan-out**: posts that link a large number of `base_id`s (hashtags, mentions,
11//!   replies) scale quadratically. Main lever against hashtag spam farms that link
12//!   hundreds of trending tags from one post.
13//! - **Bucket granularity**: submissions into tighter (shorter-duration) buckets need
14//!   more PoW than broader ones. Hot, fine-grained buckets are the most valuable
15//!   attack surface, so they get the most expensive admission control. Durations
16//!   coarser than a month (the year bucket used by Sequels) are clamped to the month
17//!   amplification so the bucket factor never drops below 1.
18//!
19//! `duration` must be one of [`crate::tools::buckets::BUCKET_DURATIONS`]; anything
20//! else is rejected with an error (this is reachable from untrusted network input,
21//! so it must never panic).
22
23use crate::tools::buckets::BUCKET_DURATIONS;
24use crate::tools::config;
25use crate::tools::time::{DurationMillis, MILLIS_IN_MONTH};
26use crate::tools::types::Pow;
27
28pub fn get_minimum_post_pow(post_length: usize, linked_base_ids_len: usize, duration: DurationMillis) -> anyhow::Result<Pow> {
29    if !BUCKET_DURATIONS.contains(&duration) {
30        anyhow::bail!("Unsupported bucket duration {}", duration);
31    }
32
33    let pow_amplification_post_length = if post_length <= 1024 { 1.0f64 } else { 1.0f64 + 1.75 * (post_length as f64 / 1024f64).log2() };
34    let pow_amplification_linked_base_ids = if linked_base_ids_len < 2 {1.0f64} else { 1.0f64 + 0.2f64 * (linked_base_ids_len as f64).powi(2) };
35    let pow_amplification_bucket = (1.0f64 + (MILLIS_IN_MONTH.0 as f64 / duration.0 as f64).log2()).max(1.0f64);
36
37    let pow_amplification = pow_amplification_post_length * pow_amplification_linked_base_ids * pow_amplification_bucket;
38    let additional_pow_bits = pow_amplification.log2().ceil();
39    Ok(Pow(config::POW_MINIMUM_PER_POST.0 + additional_pow_bits as u8))
40}

50    // baseline: small post, client_id only (< 2 links → no extra amp), month → total=1.0, ceil to +0 bits
51    #[test]
52    fn baseline() {
53        let pow = get_minimum_post_pow(512, 1, MILLIS_IN_MONTH).unwrap();
54        assert_eq!(pow, config::POW_MINIMUM_PER_POST);
55    }

58    #[test]

61        assert_eq!(pow, Pow(config::POW_MINIMUM_PER_POST.0 + 3));
62    }

120    // shorter duration → higher or equal pow
121    #[test]

123        let pow_month = get_minimum_post_pow(512, 1, MILLIS_IN_MONTH).unwrap();
124        let pow_day = get_minimum_post_pow(512, 1, MILLIS_IN_DAY).unwrap();

155    // every combination on a sample grid must be at or above the minimum
156    #[test]
157    fn never_below_minimum() {
158        let lengths = [1, 512, 1024, 4096, 64 * 1024];
159        let links = [1, 2, 5, 10];
160        for post_length in lengths {
161            for linked_base_ids_len in links {
162                for duration in BUCKET_DURATIONS {
163                    let pow = get_minimum_post_pow(post_length, linked_base_ids_len, duration).unwrap();
164                    assert!(
165                        pow >= config::POW_MINIMUM_PER_POST,
166                        "pow={:?} below minimum for post_length={} links={} duration={:?}",
167                        pow, post_length, linked_base_ids_len, duration
168                    );
169                }
170            }
171        }
172    }

190    // durations outside BUCKET_DURATIONS come from untrusted input and must be an error, never a panic
191    #[test]
192    fn rejects_unsupported_duration() {
193        let unsupported_durations = [DurationMillis(0), MILLIS_IN_SECOND.const_mul(30), MILLIS_IN_MINUTE.const_mul(2), DurationMillis(-1), -MILLIS_IN_MONTH];
194        for unsupported_duration in unsupported_durations {

199    // year buckets (Sequels) used to produce a negative bucket amplification → NaN → zero extra bits,
200    // so size and fan-out amplification were silently dropped. Year is now clamped to month.
201    #[test]
202    fn year_bucket_matches_month_bucket() {
203        let cases = [(512, 1), (64 * 1024, 1), (512, 10), (4096, 3)];
204        for (post_length, linked_base_ids_len) in cases {
205            let pow_year = get_minimum_post_pow(post_length, linked_base_ids_len, MILLIS_IN_YEAR).unwrap();
206            let pow_month = get_minimum_post_pow(post_length, linked_base_ids_len, MILLIS_IN_MONTH).unwrap();
207            assert_eq!(pow_year, pow_month, "post_length={} links={}", post_length, linked_base_ids_len);
208        }
209        assert!(get_minimum_post_pow(64 * 1024, 10, MILLIS_IN_YEAR).unwrap() > config::POW_MINIMUM_PER_POST);
210    }
