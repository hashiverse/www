1//! # Ed25519 keypair bundle with post-quantum commitment
2//!
3//! [`Keys`] packages together everything a client or server needs to sign messages and be
4//! identified on the network:
5//! - the Ed25519 secret half (`signature_key`) used to sign,
6//! - the Ed25519 public half in two forms (`verification_key` for verification, and the
7//!   serialisable `verification_key_bytes` for wire/storage),
8//! - and the 32-byte `pq_commitment_bytes` (Falcon + Dilithium commitments — see
9//!   [`crate::tools::keys_post_quantum`]) that future-proofs the identity.
10//!
11//! Construction paths:
12//! - [`Keys::from_rnd`] — generate a fresh random keypair (optionally skipping the
13//!   slow PQ commitment derivation for test scenarios).
14//! - [`Keys::from_seed`] — deterministic derivation from a 32-byte seed; used internally
15//!   and by tests that need reproducible keys.
16//! - [`Keys::from_phrase`] — Argon2 stretch a user-supplied passphrase into a 32-byte
17//!   seed, then derive keys from it. Passphrase-recoverable accounts.
18//!
19//! Persistence:
20//! - [`Keys::to_persistence`] / [`Keys::from_persistence`] encrypt the full keypair under
21//!   a passphrase using ChaCha20Poly1305 for at-rest storage in the key locker.
22

32/// Argon2id (default params) over the passphrase and raw salt bytes, producing the 32-byte
33/// ChaCha20Poly1305 key used for key persistence.  Identical output to the argon2 0.5
34/// `hash_password(passphrase, SaltString::encode_b64(salt))` path it replaced (that path
35/// decoded the salt back to raw bytes and used the default 32-byte output length) — pinned by
36/// `keys_from_persistence_regression_vector`.
37fn derive_persistence_key(passphrase: &str, salt: &[u8]) -> anyhow::Result<Key> {
38    let mut key = Key::default();
39    Argon2::default().hash_password_into(passphrase.as_bytes(), salt, &mut key).map_err(|e| anyhow!("error hashing passphrase: {}", e))?;
40    Ok(key)
41}

43#[derive(Clone)]
44pub struct Keys {
45    pub signature_key: SignatureKey,
46    pub verification_key: VerificationKey,
47    pub verification_key_bytes: VerificationKeyBytes,
48    pub pq_commitment_bytes: PQCommitmentBytes, // For a critique on this, check out https://www.reddit.com/r/cryptography/comments/1thjnaj/critique_of_a_hybrid_identity_scheme_ed25519/
49}

51impl Display for Keys {
52    fn fmt(&self, f: &mut Formatter<'_>) -> fmt::Result {
53        write!(f, "[ verification_key_bytes:{} pq_commitment:{} ]", hex::encode(self.verification_key.as_ref()), hex::encode(self.pq_commitment_bytes.as_ref()))
54    }

57impl Keys {
58    pub fn from_rnd(skip_pq_commitment_bytes: bool) -> anyhow::Result<Keys> {
59        let mut seed = [0u8; 32];
60        tools::random_fill_bytes(&mut seed);
61        Self::from_seed(&seed, skip_pq_commitment_bytes)
62    }
63
64    pub fn from_phrase(phrase: &str) -> anyhow::Result<Keys> {
65        let mut seed = [0u8; 32];
66        Argon2::default().hash_password_into(phrase.as_bytes(), b"hashiverse-global-salt", &mut seed).map_err(|e| anyhow!("error hashing phrase: {}", e))?;
67
68        Self::from_seed(&seed, false)
69    }
70
71    pub fn from_seed(seed: &[u8; 32], skip_pq_commitment_bytes: bool) -> anyhow::Result<Keys> {
72        let signature_key = {
73            let ed25519_seed = blake3::derive_key("hashiverse-pk-ed25519", seed);
74            SignatureKey::from_bytes(&ed25519_seed)?
75        };
76
77        let verification_key = signature_key.verification_key();
78        let verification_key_bytes = verification_key.to_verification_key_bytes();
79        let pq_commitment_bytes = match skip_pq_commitment_bytes {
80            false => pq_commitment_bytes_from_seed(seed),
81            true => Ok(PQCommitmentBytes::zero()),
82        }?;
83
84        Ok(Keys {
85            signature_key,
86            verification_key,
87            verification_key_bytes,
88            pq_commitment_bytes,
89        })
90    }

92    pub fn to_persistence(&self, passphrase: &String) -> anyhow::Result<String> {
93        // Concatenate key bytes into a buffer
94        let mut buf = Vec::with_capacity(32 + 32 + 32);
95        buf.extend_from_slice(self.signature_key.as_ref());
96        buf.extend_from_slice(self.verification_key.as_ref());
97        buf.extend_from_slice(self.pq_commitment_bytes.as_ref());
98
99        // Derive encryption key from passphrase and random salt
100        let mut salt = vec![0u8; 16];
101        tools::random_fill_bytes(&mut salt);
102        let key = derive_persistence_key(passphrase, &salt)?;
103
104        // Encrypt the buffer
105        let nonce = Nonce::generate();
106
107        let cipher = ChaCha20Poly1305::new(&key);
108        let ciphertext = cipher.encrypt(&nonce, buf.as_ref()).map_err(|e| anyhow!("error encrypting buffer: {}", e))?.to_vec();
109
110        // Store lengths for salt and nonce (as u8)
111        let salt_len = salt.len();
112        let nonce_len = nonce.len();
113
114        if salt_len > u8::MAX as usize || nonce_len > u8::MAX as usize {
115            return Err(anyhow!("Salt or nonce too large"));
116        }
117
118        // Prepare output: [salt_len (1)][salt bytes][nonce_len (1)][nonce bytes][ciphertext]
119        let mut out = Vec::with_capacity(1 + salt_len + 1 + nonce_len + ciphertext.len());
120        out.push(salt_len as u8);
121        out.extend_from_slice(&salt);
122        out.push(nonce_len as u8);
123        out.extend_from_slice(&nonce);
124        out.extend_from_slice(&ciphertext);
125
126        Ok(tools::encode_base64(&out))
127    }

129    pub fn from_persistence(passphrase: &String, persistence: &str) -> anyhow::Result<Keys> {
130        let decoded = tools::decode_base64(persistence)?;
131
132        if decoded.len() < 2 {
133            return Err(anyhow!("Input too short for salt/nonce lengths"));
134        }
135
136        // Get salt length and slice
137        let salt_len = decoded[0] as usize;
138        if decoded.len() < 1 + salt_len + 1 {
139            return Err(anyhow!("Input too short for salt data"));
140        }
141        let salt_start = 1;
142        let salt_end = salt_start + salt_len;
143        let salt = &decoded[salt_start..salt_end];
144
145        // Get nonce length and slice
146        let nonce_len = decoded[salt_end] as usize;
147        let nonce_start = salt_end + 1;
148        let nonce_end = nonce_start + nonce_len;
149        if decoded.len() < nonce_end {
150            return Err(anyhow!("Input too short for nonce data"));
151        }
152        let nonce = &decoded[nonce_start..nonce_end];
153        let ciphertext = &decoded[nonce_end..];
154
155        // Derive encryption key from passphrase and salt
156        let key = derive_persistence_key(passphrase, salt)?;
157
158        // Decrypt the buffer
159        let nonce = <&Nonce>::try_from(nonce).map_err(|_| anyhow!("Invalid nonce length: {}", nonce.len()))?;
160        let cipher = ChaCha20Poly1305::new(&key);
161        let buf = cipher.decrypt(nonce, ciphertext).map_err(|e| anyhow!("Decryption failed: {}", e))?;
162
163        // Split the buffer back into the key materials
164        if buf.len() != 32 * 3 {
165            return Err(anyhow!("Decrypted keys len mismatch"));
166        }
167        let signature_key_bytes = <&[u8; 32]>::try_from(&buf[0..32])?;
168        let verification_key_bytes = <&[u8; 32]>::try_from(&buf[32..64])?;
169        let pq_commitment_bytes = <&[u8; 32]>::try_from(&buf[64..96])?;
170
171        let signature_key = SignatureKey::from_bytes(signature_key_bytes)?;
172        let verification_key = VerificationKey::from_bytes_raw(verification_key_bytes)?;
173        let verification_key_bytes = verification_key.to_verification_key_bytes();
174        let pq_commitment_bytes = PQCommitmentBytes::from_slice(pq_commitment_bytes)?;
175
176        Ok(Keys {
177            signature_key,
178            verification_key,
179            verification_key_bytes,
180            pq_commitment_bytes,
181        })
182    }
183}

195        for _ in 0..8 {
196            let passphrase = Uuid::new_v4().to_string();
197
198            // Generate random keys (could also use from_seed/phrase if you want determinism)

226    #[tokio::test]
227    async fn test_falcon_sign_and_verify() -> anyhow::Result<()> {
228        use falcon_rust::falcon512;
229
230        let mut seed = [0u8; 32];
231        tools::random_fill_bytes(&mut seed);
232
233        let falcon_seed: [u8; 32] = blake3::derive_key("hashiverse-pk-falcon", &seed);
234        let (sk, pk) = falcon512::keygen(falcon_seed);
235
236        let msg = b"hello hashiverse";
237        let sig = falcon512::sign(msg, &sk);
238        assert!(falcon512::verify(msg, &sig, &pk), "Falcon signature should verify");
239
240        // Rehydrated verifying key
241        let pk_rehydrated = falcon512::PublicKey::from_bytes(&pk.to_bytes()).map_err(|e| anyhow::anyhow!("Failed to decode Falcon public key: {:?}", e))?;
242        assert!(falcon512::verify(msg, &sig, &pk_rehydrated), "Rehydrated Falcon verifying key should verify");
243
244        // Rehydrated signing key: re-serialised, signs a new message
245        let sk_rehydrated = falcon512::SecretKey::from_bytes(&sk.to_bytes()).map_err(|e| anyhow::anyhow!("Failed to decode Falcon secret key: {:?}", e))?;
246        let msg2 = b"second message";
247        let sig2 = falcon512::sign(msg2, &sk_rehydrated);
248        assert!(falcon512::verify(msg2, &sig2, &pk), "Rehydrated Falcon signing key should produce valid signatures");
249
250        // Wrong message should not verify
251        assert!(!falcon512::verify(b"wrong message", &sig, &pk), "Falcon should reject wrong message");
252
253        Ok(())
254    }
